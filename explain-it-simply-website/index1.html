// Binary search implementation
function binarySearch(arr, target) {
  let left = 0, right = arr.length - 1;
  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    if (arr[mid] === target) return mid;
    if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
  }
  return -1;
}

Explanation
Imagine you're looking for a word in a dictionary. Instead of reading every page, you open to the middle — if your word comes before that page, you throw away the second half. Keep halving the remaining pages until you find your word. That's binary search: divide and conquer, finding any item in a sorted list at incredible speed.
